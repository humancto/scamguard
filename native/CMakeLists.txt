cmake_minimum_required(VERSION 3.16)
project(scamguard_gguf_verdict LANGUAGES C CXX)

option(SCAMGUARD_BUILD_JNI "Build the Android JNI bridge" OFF)

if(NOT LLAMA_CPP_DIR)
  message(FATAL_ERROR "LLAMA_CPP_DIR must point to the pinned llama.cpp checkout")
endif()

set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)

add_subdirectory("${LLAMA_CPP_DIR}" llama.cpp-build EXCLUDE_FROM_ALL)

add_library(scamguard-gguf STATIC scamguard_gguf.cpp)
target_include_directories(scamguard-gguf PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(scamguard-gguf PUBLIC llama ggml)
target_compile_features(scamguard-gguf PUBLIC cxx_std_17)
set_target_properties(scamguard-gguf PROPERTIES POSITION_INDEPENDENT_CODE ON)

if(SCAMGUARD_BUILD_JNI)
  if(NOT ANDROID)
    message(FATAL_ERROR "SCAMGUARD_BUILD_JNI requires an Android toolchain")
  endif()
  add_library(scamguard-jni SHARED ../mobile/android/src/main/cpp/scamguard_jni.cpp)
  target_include_directories(scamguard-jni PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/include")
  target_link_libraries(scamguard-jni PRIVATE scamguard-gguf log android)
  target_compile_features(scamguard-jni PRIVATE cxx_std_17)
endif()

add_executable(scamguard-gguf-verdict gguf_verdict_runner.cpp)
target_link_libraries(scamguard-gguf-verdict PRIVATE scamguard-gguf)
target_compile_features(scamguard-gguf-verdict PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(scamguard-gguf PRIVATE /W4)
  target_compile_options(scamguard-gguf-verdict PRIVATE /W4)
else()
  target_compile_options(scamguard-gguf PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(scamguard-gguf-verdict PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(scamguard-gguf-c-abi-smoke tests/c_api_smoke.c)
  target_link_libraries(scamguard-gguf-c-abi-smoke PRIVATE scamguard-gguf)
  add_test(NAME scamguard-gguf-c-abi-smoke COMMAND scamguard-gguf-c-abi-smoke)
endif()
